local function accentInk(color)
    local function linear(v) return v<=.04045 and v/12.92 or ((v+.055)/1.055)^2.4 end
    local light=.2126*linear(color.R)+.7152*linear(color.G)+.0722*linear(color.B)
    return 1.05/(light+.05)>=4.5 and C(255,255,255) or C(10,14,20)
end

local function image(name)
    if type(name)=="number" then return "rbxassetid://"..tostring(name) end
    name = tostring(name or "layers")
    if name:match("^rbxassetid://") then return name end
    return Icons[name:gsub("^lucide%-","")] or Icons.layers or ""
end
local function validBinding(key)
    return typeof(key)=="EnumItem" and (key.EnumType==Enum.KeyCode or (key.EnumType==Enum.UserInputType and table.find({"MouseButton1","MouseButton2","MouseButton3"},key.Name)~=nil))
end
local function bindingMatches(input,key) return key.EnumType==Enum.UserInputType and input.UserInputType==key or input.KeyCode==key end
local function inputBinding(input)
    if input.KeyCode and input.KeyCode~=Enum.KeyCode.Unknown then return input.KeyCode end
    if input.UserInputType and table.find({"MouseButton1","MouseButton2","MouseButton3"},input.UserInputType.Name) then return input.UserInputType end
end

    function w:RefreshSectionSizes(onlyTab)
        if self.Destroyed or self._sizingSections then return end;self._sizingSections=true
        local sf=math.max(.01,self.Scale.Scale)
        for _,tab in ipairs(self.Tabs) do if not onlyTab or tab==onlyTab then
            for _,column in ipairs(tab.Columns) do
                local available=column.AbsoluteSize.Y/sf
                if available<=0 then available=math.max(0,height-60) end
                local sections={};for _,section in ipairs(tab.Sections) do
                    if section.Column==column and not section.Nested and section.Container.Visible then table.insert(sections,section) end
                end
                local budget=math.max(0,available-math.max(0,#sections-1)*12);local fitting={}
                for _,section in ipairs(sections) do
                    if section.AutoHeight then
                        local content=section.Layout.AbsoluteContentSize and section.Layout.AbsoluteContentSize.Y/sf or 0
                        section.DesiredHeight=math.max(0,content)+section.CaptionHeight+10
                        if section.MaxHeight then section.DesiredHeight=math.min(section.DesiredHeight,section.MaxHeight) end
                        table.insert(fitting,section)
                    else
                        local target=math.max(0,section.FixedHeight or (available*(section.HeightScale or 1)+(section.HeightOffset or 0)))
                        section.Container.Size=UDim2.new(1,0,0,target);budget=math.max(0,budget-target)
                    end
                end
                -- Water-fill: small sections keep intrinsic height; long sections share the remaining room.
                while #fitting>0 do
                    local share=budget/#fitting;local small={};local rest={}
                    for _,section in ipairs(fitting) do if section.DesiredHeight<=share then table.insert(small,section) else table.insert(rest,section) end end
                    if #small==0 then for _,section in ipairs(fitting) do section.Container.Size=UDim2.new(1,0,0,share) end;break end
                    for _,section in ipairs(small) do section.Container.Size=UDim2.new(1,0,0,section.DesiredHeight);budget=math.max(0,budget-section.DesiredHeight) end
                    fitting=rest
                end
            end
        end end
        self._sizingSections=false
    end

    end

    function w:ImportConfig(json,fireCallbacks)
        local ok,data=pcall(HttpService.JSONDecode,HttpService,json)
        if not ok or type(data)~="table" or data.Format~="SeniorAster" or data.Version~=1 or type(data.Values)~="table" then return false,"Invalid Aster config" end
        if (data.Watermark~=nil and type(data.Watermark)~="boolean") or (data.Hotkeys~=nil and type(data.Hotkeys)~="boolean") then return false,"Invalid HUD visibility" end
        if data.TweenSpeed~=nil and (type(data.TweenSpeed)~="number" or data.TweenSpeed~=data.TweenSpeed or data.TweenSpeed<0 or data.TweenSpeed>2) then return false,"Invalid tween speed" end
        if data.NotificationPosition~=nil and not table.find({"Top Left","Top Middle","Top Right","Bottom Left","Bottom Middle","Bottom Right"},data.NotificationPosition) then return false,"Invalid notification position" end
        if data.PlayerPriorities~=nil and type(data.PlayerPriorities)~="table" then return false,"Invalid player priorities" end
        for id,priority in pairs(data.PlayerPriorities or {}) do if not tostring(id):match("^%d+$") or not table.find({"Neutral","Friendly","Enemy","Priority"},priority) then return false,"Invalid player priority" end end
        local pending,alphas,bindings,palette,keymodes,animations={},{},{},{},{},{}
        local function key(name)
            local valid,value=pcall(function() return (tostring(name):match("^MouseButton[123]$") and Enum.UserInputType or Enum.KeyCode)[name] end); if valid and value and value~=Enum.KeyCode.Unknown then return value end
        end
        for flag,value in pairs(data.Values) do
            local c=self.Flags[flag]
            if c then
                if type(value)=="table" and value.Type=="Color3" then
                    if type(value.Hex)~="string" or not value.Hex:match("^%x%x%x%x%x%x$") then return false,"Invalid color" end; value=Color3.fromHex(value.Hex)
                elseif type(value)=="table" and value.Type=="KeyCode" then value=key(value.Name); if not value then return false,"Invalid key" end end
                local valid,normalized=pcall(c.Normalize,c,value); if not valid then return false,"Invalid value: "..flag end
                table.insert(pending,{c,normalized})
            end
        end
        if data.Opacity~=nil and type(data.Opacity)~="table" then return false,"Invalid opacity map" end
        for flag,value in pairs(data.Opacity or {}) do
            if self.Flags[flag] and self.Flags[flag].SetAlpha then
                if type(value)~="number" or value~=value or value<0 or value>1 then return false,"Invalid opacity" end; table.insert(alphas,{self.Flags[flag],value})
            end
        end
        if data.Bindings~=nil and type(data.Bindings)~="table" then return false,"Invalid binding map" end
        for flag,value in pairs(data.Bindings or {}) do
            if self.Flags[flag] and self.Flags[flag].SetKeybind then local k=key(value); if not k then return false,"Invalid binding" end; table.insert(bindings,{self.Flags[flag],k}) end

function Library:CreateWindow(config)
    local player=Players.LocalPlayer
    local parent=config and config.Parent or (player and player:FindFirstChild("PlayerGui"))
    local previous=parent and parent:FindFirstChild("SeniorHubAster")
    local ok,result=xpcall(function() return self:_CreateWindow(config) end,debug.traceback)
    if not ok then
        parent=parent or (player and player:FindFirstChild("PlayerGui"))
        if parent then
            local partial=parent:FindFirstChild("SeniorHubAster")
            if partial and partial~=previous then
                local dispose=partial:FindFirstChild("SeniorAsterDispose")
                if dispose then pcall(function() dispose:Invoke() end) end
                partial:Destroy()
            end
        end
        error("[Senior Aster initialization failed] "..tostring(result),0)
    end
    return result
end
